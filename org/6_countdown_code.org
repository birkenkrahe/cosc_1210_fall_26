#+begin_src python :results output :exports both :session *Python*
  def countdown(n):
      """Print n, n-1,...,1 then Liftoff!

      n -- the number to count down from (whole number)
      """
      if n <= 0:
          print("Liftoff!")
      else:
          print(n)
          countdown(n - 1) # call yourself on n-1

  countdown(3)
#+end_src

#+RESULTS:
: 3
: 2
: 1
: Liftoff!

* Countdown back and watch the unwind

#+begin_src python :results output :exports both :session *Python*
  def countdown_back(n):
      if n <= 0:  # base case
          print("Liftoff!")
      else:
          print("T-minus", n)
          countdown_back(n - 1) # call yourself on n-1
          print("...returned to", n)

  countdown_back(3)
#+end_src

  def recurse(n, s):
      if n == 0:
          print(s)
      else:
          print(n,s)  # this is the frame held during each recursive step
          recurse(n-1, n + s)

* PRACTICE Write =countdown_hold=

Write a recursive =countdown_hold(n, wind)=.

- If n <= 0 print Liftoff!
- Otherwise, if n == 5 and the wind is 25 or more: print "HOLD: wind
  too high at T-minus 5" and stop (no more calls).
- Otherwise: print T-minus n and count down from n-1.

Input: countdown_hold(8, 30)
#+begin_example python
   T-minus 8
   T-minus 7
   T-minus 6
   HOLD: wind too high at T-minus 5
#+end_example

Input: countdown_hold(8, 12)
#+begin_example python
   T-minus 8
   T-minus 7
   T-minus 6
   T-minus 5
   T-minus 4
   T-minus 3
   T-minus 2
   T-minus 1
   Liftoff!
#+end_example


#+begin_src python :results output :exports both :session *Python*
  def countdown_hold(n, wind):
        # base case met?
        if n <= 0:
              print("Liftoff!")
              # otherwise: is n == 5 and the wind stronger than 25?
        elif n == 5 and wind >= 25:
              print("HOLD: wind too high at T-minus 5")
              # otherwise: countdown (recursive case)
        else:
              print("T-minus",n)
              countdown_hold(n - 1, wind)

  countdown_hold(8, 30)
  countdown_hold(8, 12)  
#+end_src
